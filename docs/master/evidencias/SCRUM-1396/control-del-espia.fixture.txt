// CONTROL del instrumento. Tres auxiliares que crean y devuelven:
//   · `conFinally`  — el llamador borra en un finally  → con el assert fallando, debe quedar 0
//   · `sinFinally`  — el llamador borra DESPUÉS del assert, sin finally → con el fallo, debe QUEDAR
//   · `nunca`       — nadie borra (sólo si CONTROL_FUGA) → queda siempre
import test from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';

function conFinally() { const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'control-fin-')); return { dir }; }
function sinFinally() { const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'control-sin-')); return { dir }; }
function nunca() { const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'control-nunca-')); return { dir }; }

test('con finally', () => { const b = conFinally(); try { assert.equal(1, 1); } finally { fs.rmSync(b.dir, { recursive: true, force: true }); } });
test('sin finally', () => { const b = sinFinally(); assert.equal(1, 1); fs.rmSync(b.dir, { recursive: true, force: true }); });
if (process.env.CONTROL_FUGA) test('nunca', () => { const b = nunca(); assert.ok(b.dir); });
