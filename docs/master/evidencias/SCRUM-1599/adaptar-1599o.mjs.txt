// SOLO PARA MEDIR (SCRUM-1599o): adapta EN LOCAL los 11 con carrera para que corran contra LIBRO_PG_URL.
// No se comitea: se deshace con `git restore tests/`. Uso: node adaptar.mjs <raiz>
// Forma staging (7): la de 1599m, import de _banco-libro detrás de _staging-db + `|| URL_BANCO !== ''`.
// Forma dev (4): la clave DATABASE_URL_DEV se lee de LIBRO_PG_URL y la base exigida pasa a yaqu_libro_test.
import fs from 'node:fs';
const raiz = process.argv[2];
const STAGING = ['scrum115-wa-fallo-registrado', 'scrum127-paywall-bloquea', 'scrum173-cadena-verifactu-serializada',
  'scrum178-emision-manual', 'scrum234-carrera-serie.gated', 'scrum814-carrera-del-tramo.gated', 'scrum92-login-operario'];
const DEV = ['scrum592-concurrencia-serie', 'scrum767-cura-concurrencia', 'scrum781-concurrencia-de-la-factura', 'scrum793-la-carrera-del-token'];
let mal = 0;
const cambia = (t, de, a, nombre, f) => {
  if (!de.test(t)) { console.log(`NO CASA ${nombre} en ${f}`); mal++; return t; }
  return t.replace(de, a);
};
for (const f of STAGING) {
  const p = `${raiz}/tests/${f}.test.mjs`;
  let t = fs.readFileSync(p, 'utf8');
  t = cambia(t, /^(import '\.\/_staging-db\.mjs';[^\n]*\n)/m, `$1import { URL_BANCO } from './_banco-libro.mjs';\n`, 'import', f);
  t = cambia(t, /^const ENABLED = process\.env\.QA_DB_TEST === '1';/m, `const ENABLED = process.env.QA_DB_TEST === '1' || URL_BANCO !== '';`, 'gate', f);
  fs.writeFileSync(p, t);
}
for (const f of DEV) {
  const p = `${raiz}/tests/${f}.test.mjs`;
  let t = fs.readFileSync(p, 'utf8');
  t = cambia(t, /^const (ENABLED|GATE) = process\.env\.QA_DB_TEST === '1';/m, `const $1 = process.env.QA_DB_TEST === '1' || !!process.env.LIBRO_PG_URL;`, 'gate', f);
  t = cambia(t, /^(\s*)const linea = [^\n]*DATABASE_URL_DEV=[^\n]*$/m, `$1const linea = 'DATABASE_URL_DEV=' + process.env.LIBRO_PG_URL;`, 'linea', f);
  t = cambia(t, /info\.base === 'yaqu_dev_javier'/, `info.base === 'yaqu_libro_test'`, 'base', f);
  fs.writeFileSync(p, t);
}
console.log(`adaptados=${STAGING.length + DEV.length} sin_casar=${mal}`);
process.exit(mal ? 1 : 0);
