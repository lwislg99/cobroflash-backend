// SCRUM-1599r · UNA mutación en dist/ (nunca en src/: invoiceNumber.service.ts es camino de emisión, regla 38).
// Rompe el formato F: la rama `usaFormatoF` de formatInvoiceNumber no se toma nunca. Exige ancla con recuento 1,
// corre los ficheros con correr.mjs y restaura dist comprobando el sha256. Uso: node mutar-1599r.mjs <raiz> <puerto> <f…>
import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
const [raiz, puerto, ...ficheros] = process.argv.slice(2);
const AQUI = path.dirname(fileURLToPath(import.meta.url));
const DIST = path.join(raiz, 'dist/modules/invoicing/domain/invoiceNumber.service.js');
const DE = '    if (usaFormatoF(emitidaEn, rectifying, corte)) {';
const A = '    if (false) { // MUTACIÓN 1599r';
const sha = (b) => crypto.createHash('sha256').update(b).digest('hex');
const original = fs.readFileSync(DIST);
const h0 = sha(original);
const veces = original.toString('utf8').split(DE).length - 1;
console.log(`ancla=${JSON.stringify(DE.trim())} veces=${veces} sha0=${h0.slice(0, 12)}`);
if (veces !== 1) { console.log('CIEGO: el ancla no sale UNA vez'); process.exit(2); }
let rc;
try {
  fs.writeFileSync(DIST, original.toString('utf8').replace(DE, A));
  console.log(`mutado sha=${sha(fs.readFileSync(DIST)).slice(0, 12)}`);
  const r = spawnSync(process.execPath, [path.join(AQUI, 'correr.mjs'), raiz, puerto, 'MUTACION', '-', ...ficheros], { encoding: 'utf8' });
  process.stdout.write(r.stdout + r.stderr);
  rc = r.status;
} finally {
  fs.writeFileSync(DIST, original);
  const h1 = sha(fs.readFileSync(DIST));
  console.log(`restaurado sha=${h1.slice(0, 12)} igual=${h1 === h0}`);
  if (h1 !== h0) process.exit(3);
}
console.log(`correr rc=${rc} (1 = algún fichero cae: la mutación MUERE)`);
