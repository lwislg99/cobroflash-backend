// SCRUM-1599q · corre ficheros en serie contra el Postgres desechable, uno por `node --test`, y cuenta casos
// anclados al inicio de línea (primer nivel) que tienen que sumar el `# tests` declarado. Sin casos → CIEGO.
// Uso: node correr.mjs <raiz> <puerto> <etiqueta> <reloj ISO | -> <tests/f…>
// Con reloj: NODE_OPTIONS=--import reloj.mjs y una MARCA por proceso; sin marca, la sonda del reloj es CIEGA.
import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';
import { pathToFileURL, fileURLToPath } from 'node:url';

const [raiz, puerto, etiqueta, reloj, ...ficheros] = process.argv.slice(2);
const AQUI = path.dirname(fileURLToPath(import.meta.url));
const CASO = /^(\s*)(ok|not ok) \d+ - (.*?)(?: # (SKIP|TODO)\b.*)?$/;
const entorno = { ...process.env };
for (const k of ['NODE_TEST_CONTEXT', 'NODE_OPTIONS', 'FORCE_COLOR', 'QA_DB_TEST', 'A55_DB_TEST', 'BOT_SUITE_TEST', 'DATABASE_URL', 'DATABASE_URL_TESTS', 'DATABASE_URL_DEV']) delete entorno[k];
let rc = 0;
for (const f of ficheros) {
  const tap = path.join(AQUI, `tap-${etiqueta}-${path.basename(f)}.tap`);
  const marca = path.join(AQUI, `marca-${etiqueta}-${path.basename(f)}.txt`);
  fs.rmSync(tap, { force: true }); fs.rmSync(marca, { force: true });
  const env = { ...entorno, LIBRO_PG_URL: `postgresql://postgres@127.0.0.1:${puerto}/yaqu_libro_test`, DISABLE_CRONS: 'true', WHATSAPP_DRY_RUN: '1', NODE_ENV: 'test' };
  if (reloj !== '-') { env.RELOJ_1599Q = reloj; env.RELOJ_1599Q_MARCA = marca; env.NODE_OPTIONS = `--import=${pathToFileURL(path.join(AQUI, 'reloj.mjs')).href}`; }
  const t0 = Date.now();
  const r = spawnSync(process.execPath, ['--test', '--test-force-exit', '--test-reporter=tap', `--test-reporter-destination=${tap}`, f], { cwd: raiz, env, encoding: 'utf8', timeout: 150_000 });
  const lineas = fs.existsSync(tap) ? fs.readFileSync(tap, 'utf8').split(/\r?\n/) : [];
  let pasan = 0, caen = 0, saltan = 0, decl = null; const caidos = [];
  for (const l of lineas) {
    const d = /^# tests (\d+)$/.exec(l); if (d) decl = Number(d[1]);
    const m = CASO.exec(l); if (!m || m[1].length > 0) continue;
    if (m[4] === 'SKIP') saltan++; else if (m[2] === 'ok') pasan++; else { caen++; caidos.push(m[3]); }
  }
  const casos = pasan + caen + saltan;
  const marcas = fs.existsSync(marca) ? fs.readFileSync(marca, 'utf8').trim().split('\n').filter(Boolean) : [];
  const relojVisto = reloj === '-' ? 'sin-reloj' : marcas.length ? `reloj-cargado×${marcas.length} (${marcas[0].split('ahora=')[1]})` : 'RELOJ-CIEGO';
  const v = casos === 0 ? 'CIEGO' : (reloj !== '-' && !marcas.length) ? 'CIEGO' : caen > 0 || r.status !== 0 ? 'ROJO' : saltan > 0 ? 'SALTA' : 'VERDE';
  if (v !== 'VERDE') rc = 1;
  console.log(`${etiqueta} · ${v.padEnd(5)} · ${Date.now() - t0} ms · ${pasan}/${caen}/${saltan} (decl ${decl}, suman ${decl === casos}) · ${relojVisto} · ${path.basename(f)}${caidos.length ? ' · CAEN: ' + caidos.join(' | ') : ''}`);
  // El mensaje de la aserción que cae, literal (para decir QUÉ cae, no sólo que cae).
  for (const l of lineas) if (/^\s+(error|expected|actual):/.test(l)) console.log(`    ${l.trim().slice(0, 300)}`);
}
process.exit(rc);
