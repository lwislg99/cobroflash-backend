// SCRUM-1599t · el VECINO de 574: lo que deja otro test del banco a mitad de su ejecución.
// Uso: node vecino-574.mjs <raiz> <puerto> <etiqueta> <EMPRESA|ninguno> <tests/f…>
// Dentro de un merchant efímero (withMerchant, que lo borra al salir) crea UN cliente con la forma
// jurídica dada —lo mismo que hace el control positivo de 574 en su vuelta por EMPRESA— y, con él
// vivo, corre el fichero con el corredor de 1599q. «ninguno» crea el cliente sin declarar: es el
// gemelo que NO debería tumbar el caso, y separa «hay un vecino» de «el vecino está declarado».
import path from 'node:path';
import { spawnSync } from 'node:child_process';
import { pathToFileURL, fileURLToPath } from 'node:url';

const [raiz, puerto, etiqueta, forma, fichero] = process.argv.slice(2);
process.env.DATABASE_URL = `postgresql://postgres@127.0.0.1:${puerto}/yaqu_libro_test`;
const { prisma } = await import(pathToFileURL(path.join(raiz, 'dist/core/db/prisma.js')).href);
const { withMerchant } = await import(pathToFileURL(path.join(raiz, 'tests/_merchant-fixture.mjs')).href);

await withMerchant(prisma, { name: `QA vecino 1599t ${Date.now()}` }, async (m) => {
  const c = await prisma.customer.create({
    data: { merchantId: m.id, name: 'Vecino 1599t', contactKind: forma === 'ninguno' ? null : forma },
  });
  const n = await prisma.customer.count();
  console.log(`VECINO merchant=${m.id} cliente=${c.id} contactKind=${c.contactKind} · clientes en la base=${n}`);
  const r = spawnSync(process.execPath, [path.join(path.dirname(fileURLToPath(import.meta.url)), 'correr.mjs'), raiz, puerto, etiqueta, '-', fichero], { cwd: raiz, encoding: 'utf8' });
  process.stdout.write(r.stdout); process.stderr.write(r.stderr);
});
console.log(`TRAS LIMPIAR · clientes en la base=${await prisma.customer.count()}`);
await prisma.$disconnect();
