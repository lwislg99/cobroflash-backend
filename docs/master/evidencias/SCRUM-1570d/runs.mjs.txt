// Lista los runs de CI (push a main) en una ventana y, por cada uno, el job de navegador y sus anotaciones.
import { execFileSync } from 'node:child_process';
import { writeFileSync } from 'node:fs';
const gh = (a) => execFileSync('gh', a, { encoding: 'utf8', maxBuffer: 1 << 28 });
const [desde, hasta, out] = process.argv.slice(2);
const runs = [];
for (let page = 1; page <= 10; page++) {
  const r = JSON.parse(gh(['api', `repos/lwislg99/cobroflash-backend/actions/workflows/ci.yml/runs?branch=main&event=push&per_page=100&page=${page}&created=${desde}..${hasta}`]));
  runs.push(...r.workflow_runs);
  if (r.workflow_runs.length < 100) break;
}
const filas = [];
for (const run of runs) {
  const jobs = JSON.parse(gh(['api', `repos/lwislg99/cobroflash-backend/actions/runs/${run.id}/jobs?per_page=100`])).jobs;
  const j = jobs.find((x) => x.name === 'guards de navegador (fuera de la tanda)');
  let anot = [];
  if (j && j.check_run_url) {
    const cid = j.check_run_url.split('/').pop();
    anot = JSON.parse(gh(['api', `repos/lwislg99/cobroflash-backend/check-runs/${cid}/annotations?per_page=100`]));
  }
  filas.push({ run: run.id, sha: run.head_sha, created: run.created_at, runConcl: run.conclusion,
    job: j ? j.id : null, jobConcl: j ? j.conclusion : null, jobsTotal: jobs.length,
    pasos: j ? j.steps.map((s) => `${s.name}=${s.conclusion}`) : [],
    anot: anot.map((a) => `${a.annotation_level}|${a.title}|${a.message}`) });
}
writeFileSync(out, JSON.stringify(filas, null, 1));
console.log(`runs=${runs.length} conJob=${filas.filter((f) => f.job).length}`);
