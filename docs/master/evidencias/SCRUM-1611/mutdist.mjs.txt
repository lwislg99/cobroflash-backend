// SCRUM-1611 · muta UNA línea de un fichero de dist/ (literal exacto, una sola aparición) o lo restaura por sha256.
// Uso: node mutdist.mjs mutar <fichero dist> <de> <a>   |   node mutdist.mjs restaurar <fichero dist> <sha256 esperado>
// La copia original va a <fichero>.orig1611; restaurar la devuelve y comprueba el sha256.
import fs from 'node:fs';
import crypto from 'node:crypto';
const [orden, f, x, y] = process.argv.slice(2);
const sha = (p) => crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
if (orden === 'mutar') {
  const t = fs.readFileSync(f, 'utf8');
  const n = t.split(x).length - 1;
  if (n !== 1) { console.log(`CIEGO: el literal aparece ${n} veces`); process.exit(2); }
  if (!fs.existsSync(f + '.orig1611')) fs.copyFileSync(f, f + '.orig1611');
  fs.writeFileSync(f, t.replace(x, y));
  console.log(`MUTADO · ${f} · sha ${sha(f).slice(0, 16)}`);
} else if (orden === 'restaurar') {
  fs.copyFileSync(f + '.orig1611', f);
  const s = sha(f);
  if (s !== x) { console.log(`🔴 sha distinto tras restaurar: ${s}`); process.exit(1); }
  fs.rmSync(f + '.orig1611');
  console.log(`RESTAURADO · ${f} · sha ${s.slice(0, 16)} = esperado`);
} else { console.log('CIEGO: orden'); process.exit(2); }
