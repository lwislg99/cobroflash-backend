const filas = [];
async function fila(sitio, como, fn) {
  const r = {};
  for (const k of Object.keys(DOCS)) {
    try { r[k] = await fn({ ...DOCS[k] }); } catch (e) { r[k] = 'LANZA ' + (e && (e.name !== 'Error' ? e.name + ': ' : '') + (e && e.message)); }
  }
  // Se compara con el número de cada documento sustituido por <N>: «Tu factura F260001» y
  // «Tu factura J-2026-0001» dicen lo mismo de su documento.
  const sin = (k) => JSON.stringify(r[k]).split(DOCS[k].number).join('<N>');
  const ciega = sin('F1') === sin('JUST');
  const lado = ciega ? 'CIEGA (F1 = JUST)' : sin('F1J') === sin('F1') ? 'como FACTURA' : sin('F1J') === sin('JUST') ? 'como JUSTIFICANTE' : 'NI UNA NI OTRA';
  filas.push({ sitio, como, lado, ...r });
}

// Texto visible de un PDF de pdfkit con fuentes estándar: se inflan los streams y se decodifican
// los `<hex> Tj/TJ`. Control: el JUST tiene que dar «JUSTIFICANTE DE COBRO», o el lector es ciego.
function textoDelPdf(buf) {
  const s = buf.toString('latin1'); const out = [];
  const re = /stream\r?\n([\s\S]*?)\r?\nendstream/g; let m;
  while ((m = re.exec(s))) {
    let plano; try { plano = zlib.inflateSync(Buffer.from(m[1], 'latin1')).toString('latin1'); } catch { continue; }
    const hx = /<([0-9a-fA-F]+)>/g; let h;
    while ((h = hx.exec(plano))) out.push(Buffer.from(h[1], 'hex').toString('latin1'));
  }
  return out.join('');
}
