const capaDe = (ruta, m) => quoteDecisionLandingRouter.stack.find((l) => l.route && [].concat(l.route.path).includes(ruta) && l.route.methods[m]);
async function pedir(ruta, m, req) {
  const capa = capaDe(ruta, m);
  if (!capa) throw new Error('CIEGO: sin ruta ' + m + ' ' + ruta);
  let cuerpo = '', st = 200;
  const res = { status(s) { st = s; return res; }, setHeader() { return res; }, redirect(u) { cuerpo = 'REDIRECT ' + u; return res; }, send(b) { cuerpo = String(b); return res; } };
  const req2 = { params: { token: TOKEN }, query: {}, headers: {}, ip: '127.0.0.1', socket: {}, body: {}, ...req };
  await capa.route.stack.at(-1).handle(req2, res);
  return { st, cuerpo };
}
