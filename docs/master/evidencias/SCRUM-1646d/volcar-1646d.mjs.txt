process.env.WHATSAPP_DRY_RUN = '1';
import fs from 'node:fs';
import path from 'node:path';
import http from 'node:http';
import { pathToFileURL } from 'node:url';
const [ARBOL, OUT] = process.argv.slice(2);
const api = http.createServer((q, r) => { r.setHeader('content-type', 'application/json'); r.end('{"ok":true}'); });
await new Promise((ok) => api.listen(0, '127.0.0.1', ok));
process.env.PUBLIC_BASE_URL = 'http://127.0.0.1:' + api.address().port;
const T = (p) => pathToFileURL(path.join(ARBOL, p)).href;
const { inyectarBase, moduloDeDist, MERCHANT, CLIENTE } = await import(T('tests/_envio-doblado.mjs'));
const LANDING = '../dist/modules/system/app/routes/quoteDecisionLanding.routes.js';
const TOKEN = 'c'.repeat(32);
const FIJA = '2026-11-12T10:00:00.000Z';
const NEGOCIO = "Fontanería & Hijos <Sur> 'Pepe'";
const CLIENTE_NOMBRE = 'Cliente Ñoño & <b>1646</b>';
const MARCA = '#C2410c'; // mayúsculas y minúsculas: el filtro las acepta y la hoja las copia tal cual
let FILA = null;
inyectarBase({
  'quote.findUnique': ({ where }) => (where.decisionToken === TOKEN ? JSON.parse(JSON.stringify(FILA)) : null),
  'quote.findFirst': ({ where }) => (where.decisionToken === TOKEN ? JSON.parse(JSON.stringify(FILA)) : null),
  'quote.count': () => 0,
}, [LANDING]);
const { quoteDecisionLandingRouter } = moduloDeDist(LANDING);
const opcion = (id, label, precio, recommended) => ({
  id, label, description: '', recommended, total: (precio * 1.21).toFixed(2),
  lines: [{ concept: `Trabajo ${label}`, qty: 1, price: precio, tax: 0.21 }],
});
const TRES = [opcion('basica', 'Básica', 100, false), opcion('completa', 'Completa & más', 200, true), opcion('total', 'Total', 300, false)];
function fila(extra = {}, marca = null) {
  return {
    id: 1646, merchantId: MERCHANT, customerId: CLIENTE, status: 'sent', total: '121.00', currency: 'EUR',
    lines: [{ concept: 'Revisión de caldera', qty: 1, price: 100, tax: 0.21 }], quoteNumber: 1646, revision: 0, seriesYear: 2026,
    decisionToken: TOKEN, validUntil: '2027-01-31T10:00:00.000Z', paymentTerms: 'FULL_UPFRONT',
    tiers: null, discountGlobalAmount: null, acceptedAt: null, rejectedAt: null, rejectionReason: null, jobId: null,
    createdAt: FIJA, updatedAt: FIJA, signatureUrl: null,
    merchant: { id: MERCHANT, name: NEGOCIO, legalName: null, taxId: null, country: 'ES', email: null,
      whatsappPhone: '+34600000000', defaultCurrency: 'EUR', flags: null, timezone: 'Europe/Madrid', brandColor: marca,
      brandAccentColor: null, logoUrl: null, address: null, clausulasPresupuesto: null },
    customer: { id: CLIENTE, name: CLIENTE_NOMBRE },
    Invoice: [],
    ...extra,
  };
}
const capaDe = (ruta, m) => quoteDecisionLandingRouter.stack.find((l) => l.route && [].concat(l.route.path).includes(ruta) && l.route.methods[m]);
async function pedir(ruta, m, req) {
  const capa = capaDe(ruta, m);
  if (!capa) throw new Error('CIEGO: sin ruta ' + m + ' ' + ruta);
  let cuerpo = '', st = 200;
  const res = { status(s) { st = s; return res; }, setHeader() { return res; }, redirect(u) { cuerpo = 'REDIRECT ' + u; return res; }, send(b) { cuerpo = String(b); return res; } };
  const req2 = { params: { token: TOKEN }, query: {}, headers: {}, ip: '127.0.0.1', socket: {}, body: {}, ...req };
  await capa.route.stack.at(-1).handle(req2, res);
  return { st, cuerpo };
}
const GET = () => pedir('/quote/:token', 'get', {});
const CARAS = {
  uno: [GET, {}], tres: [GET, { tiers: TRES, total: '242.00' }],
  aceptado: [GET, { status: 'accepted', acceptedAt: FIJA }],
  rechazado: [GET, { status: 'rejected', rejectedAt: FIJA, rejectionReason: 'El precio es demasiado alto' }],
  caducado: [GET, { validUntil: '2026-01-31T10:00:00.000Z' }],
  'rechazo-form': [() => pedir('/quote/:token/reject', 'get', {}), {}],
  'rechazo-enviado': [() => pedir('/quote/:token/reject', 'post', { body: { reason: 'price', comment: 'Caro' } }), {}],
};
fs.mkdirSync(OUT, { recursive: true });
for (const [base, [f, extra]] of Object.entries(CARAS)) {
  for (const marca of [null, MARCA]) {
    const n = marca ? base + '-marca' : base;
    FILA = fila(extra, marca);
    const { st, cuerpo } = await f();
    if (st !== 200 || cuerpo.length < 500) throw new Error('CIEGO ' + n + ' ' + st + ' ' + cuerpo.slice(0, 80));
    if (/Ã|Â/.test(cuerpo)) throw new Error('CIEGO: tildes rotas en ' + n);
    if ((base === 'uno' || base === 'tres') && !cuerpo.includes('Fontanería')) throw new Error('CIEGO: sin el nombre del negocio en ' + n);
    if (base === 'caducado' && !cuerpo.includes('caducó')) throw new Error('CIEGO: la cara de caducado no es la de caducado');
    if (base === 'uno' && !cuerpo.includes('terms-badge')) throw new Error('CIEGO: sin condiciones de pago');
    if (base === 'uno' && !cuerpo.includes('validity-badge')) throw new Error('CIEGO: sin validez');
    fs.writeFileSync(path.join(OUT, n + '.html'), cuerpo);
    const estilos = (cuerpo.match(/<style\b/gi) || []).length;
    const attrs = (cuerpo.match(/\sstyle\s*=/gi) || []).length;
    const hojas = (cuerpo.match(/<link\b[^>]*rel="stylesheet"[^>]*>/g) || []).map((h) => /href="([^"]*)"/.exec(h)[1]);
    console.log(n.padEnd(22), String(cuerpo.length).padStart(6), 'bytes · <style>:', estilos, '· style=:', attrs, '· hojas:', hojas.join(' '));
  }
}
process.exit(0);
