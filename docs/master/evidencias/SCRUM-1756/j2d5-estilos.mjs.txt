const TIPOS = { '.html': 'text/html; charset=utf-8', '.js': 'application/javascript', '.css': 'text/css', '.svg': 'image/svg+xml', '.png': 'image/png', '.json': 'application/json', '.webmanifest': 'application/manifest+json' };
function servidor(raiz, conPolitica) {
  return http.createServer((req, res) => {
    if (conPolitica) res.setHeader('Content-Security-Policy', POLITICA);
    const p = decodeURIComponent(new URL(req.url, 'http://x').pathname);
    if (p.startsWith('/admin/')) { res.setHeader('Content-Type', 'application/json'); res.end(JSON.stringify(api(req))); return; }
    if (p === '/__sembrado') { res.setHeader('Content-Type', 'text/html; charset=utf-8'); res.end('<!doctype html><p id="s" style="color:rgb(1, 2, 3)">x</p>'); return; }
    if (p === '/sw.js') { res.statusCode = 404; res.end(); return; } // sin service worker: no cachea entre corridas
    const f = path.join(raiz, p.endsWith('/') ? p + 'index.html' : p);
    if (!f.startsWith(raiz) || !fs.existsSync(f) || fs.statSync(f).isDirectory()) { res.statusCode = 404; res.end(); return; }
    res.setHeader('Content-Type', TIPOS[path.extname(f)] || 'application/octet-stream');
    res.end(fs.readFileSync(f));
  });
}

const RECOGER = () => {

  const filas = [];
  for (const r of raices) for (const el of [r, ...r.querySelectorAll('*')]) {
    const cs = getComputedStyle(el);
    const o = {};
    for (let i = 0; i < cs.length; i++) o[cs[i]] = cs.getPropertyValue(cs[i]);
    const id = el.id ? '#' + el.id : '';
    filas.push({ tag: el.tagName + id, clase: el.getAttribute('class') || '', estilos: o });
  }
  return filas;
};

async function correr(clave, esc, ancho) {
  variante = esc.variante || 'lleno';
  const page = await navegador.newPage();
  await page.setViewport({ width: ancho, height: 900 });
  await page.emulateMediaFeatures([{ name: 'prefers-reduced-motion', value: 'reduce' }]);
  const errores = [];
  page.on('pageerror', (e) => errores.push(String(e.message).slice(0, 160)));
  await page.evaluateOnNewDocument(() => {
    window.__violaciones = [];
    document.addEventListener('securitypolicyviolation', (e) => window.__violaciones.push({ dir: e.effectiveDirective, src: e.sourceFile, linea: e.lineNumber }));
  });
  await page.goto(`http://127.0.0.1:${PUERTOS[clave]}/dashboard/index.html${esc.hash}`, { waitUntil: 'networkidle0' });
  await pausa(400);
  for (const paso of esc.pasos) {
    if (paso[0] === 'click') {
      if (paso[2] === 'importar') {
        const ok = await page.evaluate(() => { const b = [...document.querySelectorAll('#view-container button')].find((x) => /importar/i.test(x.textContent)); if (b) b.click(); return !!b; });
        if (!ok) errores.push('no encontré el botón de importar');
      } else {
        const ok = await page.evaluate((s) => { const b = document.querySelector(s); if (b) b.click(); return !!b; }, paso[1]);
        if (!ok) errores.push('no encontré ' + paso[1]);
      }
    } else if (paso[0] === 'fichero') {
      const input = await page.$('#csv-file-input');
      if (!input) errores.push('no encontré #csv-file-input'); else await input.uploadFile(CSV);
    }
    await pausa(350);
  }
  const filas = await page.evaluate(RECOGER);
  const viol = await page.evaluate(() => window.__violaciones);
  await page.close();
  return { filas, viol, errores };
}
function comparar(a, b) {
  const difs = [];
  if (a.filas.length !== b.filas.length) return { estructura: `${a.filas.length} ≠ ${b.filas.length} elementos`, difs };
  for (let i = 0; i < a.filas.length; i++) {
    const x = a.filas[i], y = b.filas[i];
    if (x.tag !== y.tag) return { estructura: `elemento ${i}: ${x.tag} ≠ ${y.tag}`, difs };
    for (const k of Object.keys(x.estilos)) if (x.estilos[k] !== y.estilos[k]) difs.push(`${i} ${x.tag}.${(y.clase || '').replace(/ /g, '.')} ${k}: «${x.estilos[k]}» → «${y.estilos[k]}»`);
  }
  return { estructura: null, difs };
}
