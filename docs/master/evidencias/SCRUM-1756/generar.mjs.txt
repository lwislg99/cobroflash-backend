function veces(hay, aguja) { return hay.split(aguja).length - 1; }
function exacto(hay, aguja, n, donde) {
  const v = veces(hay, aguja);
  if (v !== n) throw new Error(`${donde}: «${aguja.slice(0, 80)}» aparece ${v} veces (se esperaban ${n})`);
}
function leer(f) {
  const s = fs.readFileSync(f, 'utf8');
  if (s.includes('\r')) throw new Error(`${f}: lleva CR`);
  return s;
}
// Saca el CSS entre `abre` y `cierra` (los dos únicos), y pone `pon` en el hueco de los tres.
function sacar(src, abre, cierra, pon, donde) {
  exacto(src, abre, 1, donde); exacto(src, cierra, 1, donde);
  const i = src.indexOf(abre); const j = src.indexOf(cierra);
  if (!(i < j)) throw new Error(`${donde}: orden de <style>`);
  const css = src.slice(i + abre.length, j);
  if (css.includes('${') || css.includes('`')) throw new Error(`${donde}: el CSS lleva \${ o comilla grave`);
  if (/<\/?style/i.test(css)) throw new Error(`${donde}: el CSS lleva otra etiqueta <style>`);
  return { src: src.slice(0, i) + pon + src.slice(j + cierra.length), css };
}
function cambiar(src, antes, despues, n, donde) {
  exacto(src, antes, n, donde);
  return src.split(antes).join(despues);
}
