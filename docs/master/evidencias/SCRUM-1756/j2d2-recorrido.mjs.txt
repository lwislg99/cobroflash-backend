function entornoHijo(puerto) {
  const e = { ...process.env, DISABLE_CRONS: 'true', NODE_ENV: 'test', MP_ACCESS_TOKEN: '' };
  for (const k of Object.keys(e)) if (/^DATABASE_URL/.test(k) || k === 'NODE_TEST_CONTEXT' || k === 'NODE_OPTIONS') delete e[k];

  e.NODE_PATH = path.join(RAIZ, 'node_modules');

}
function arrancar(esc) {
  return new Promise((ok, mal) => {
    const h = spawn(process.execPath, [SERVIDOR, esc.dist, String(esc.puerto), esc.modo],
      { cwd: path.dirname(path.resolve(esc.dist)), env: entornoHijo(esc.puerto), stdio: ['ignore', 'pipe', 'pipe'] });
    let log = '';
    const t = setTimeout(() => mal(new Error(`${esc.id}: sin LISTO en 60 s\n${log}`)), 60_000);
    h.stdout.on('data', (d) => { log += d; if (log.includes('LISTO')) { clearTimeout(t); ok(h); } });
    h.stderr.on('data', (d) => { log += d; });
    h.on('exit', (c) => { clearTimeout(t); mal(new Error(`${esc.id}: salió con ${c}\n${log.slice(-800)}`)); });
  });
}

async function abrir(nav, url) {
  const ctx = await nav.createBrowserContext();
  const p = await ctx.newPage();
  const fuera = [];
  p.on('request', (q) => { if (!q.url().startsWith('http://127.0.0.1:') && !q.url().startsWith('data:')) fuera.push(q.url()); });
  p.on('dialog', async (d) => { fuera.push(`DIALOGO ${d.message()}`); await d.dismiss(); });

  await p.setViewport({ width: 390, height: 900, deviceScaleFactor: 1 });
  const r = await p.goto(url, { waitUntil: 'networkidle0' });
  return { ctx, p, status: r.status(), fuera };
}
