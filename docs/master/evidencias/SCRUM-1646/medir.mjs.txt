function censo(html) {
  const sinComentarios = html.replace(/<!--[\s\S]*?-->/g, '');
  const scripts = [...sinComentarios.matchAll(/<script\b([^>]*)>([\s\S]*?)<\/script>/gi)];
  const enLinea = scripts.filter((m) => !/\bsrc=/.test(m[1]) && !/type="application\/json"/.test(m[1]));
  const etiquetas = [...sinComentarios.matchAll(/<[a-zA-Z][^>]*>/g)].map((m) => m[0]);
  return {
    scriptEnLinea: enLinea.length,
    on: etiquetas.filter((t) => /\son[a-z]+\s*=/i.test(t)).length,
    styleEtiqueta: (sinComentarios.match(/<style\b/gi) || []).length,
    styleAtributo: etiquetas.filter((t) => /\sstyle\s*=/i.test(t)).length,
    // dentro de cadenas JS que acaban en innerHTML (no son etiquetas del documento, pero nacerán en él)
    styleEnCadenasJs: enLinea.reduce((n, m) => n + (m[2].match(/style=\\?"/g) || []).length, 0),
    innerHTML: enLinea.reduce((n, m) => n + (m[2].match(/\.innerHTML\s*=/g) || []).length, 0),
  };
}
