// SCRUM-1631 · Postgres 16 desechable en loopback. Forma de arrancar-pg-1599t.mjs.txt (UTF8, DDL de migrate diff --from-empty).
// Uso: node arrancar-pg-1631.mjs <dir> <puerto> <ddl.sql>. Se queda vivo hasta SIGINT/kill.
import EmbeddedPostgres from 'embedded-postgres';
import fs from 'node:fs';
import pg from 'pg';

const [dir, puertoTxt, ddl] = process.argv.slice(2);
const port = Number(puertoTxt);
fs.rmSync(dir, { recursive: true, force: true });
const db = new EmbeddedPostgres({ databaseDir: dir, user: 'postgres', password: 'x', port, persistent: false, authMethod: 'trust', initdbFlags: ['--encoding=UTF8', '--locale=C'] });
await db.initialise();
await db.start();
await db.createDatabase('yaqu_libro_test');
const c = new pg.Client({ host: '127.0.0.1', port, user: 'postgres', database: 'yaqu_libro_test' });
await c.connect();
await c.query(fs.readFileSync(ddl, 'utf8'));
const r = await c.query("select count(*)::int n from information_schema.tables where table_schema='public'");
const v = await c.query('show server_version');
const e = await c.query('show server_encoding');
await c.end();
console.log(`LISTO puerto=${port} tablas=${r.rows[0].n} version=${v.rows[0].server_version} encoding=${e.rows[0].server_encoding}`);
const parar = async () => { await db.stop(); process.exit(0); };
process.on('SIGINT', parar);
process.on('SIGTERM', parar);
setInterval(() => {}, 1 << 30);
