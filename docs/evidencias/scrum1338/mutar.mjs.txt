// SCRUM-1338 · mutaciones de FRONT: cada una estropea el producto y el guard tiene que caer.
//   node mutar-1338.mjs <raiz ABSOLUTA del repo>
// Base sin mutar primero. Tras cada mutación: `git restore --source=HEAD` y árbol limpio comprobado.
import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const RAIZ = path.resolve(process.argv[2]);
const TEST = 'tests/scrum1338-la-barra-del-operario-contra-los-gates.test.mjs';
const entorno = { ...process.env };
for (const k of ['FORCE_COLOR', 'NODE_OPTIONS', 'NODE_TEST_CONTEXT']) delete entorno[k];
const git = (...a) => spawnSync('git', a, { cwd: RAIZ, encoding: 'utf8' });

function correr() {
  const r = spawnSync(process.execPath, ['--test', '--test-reporter=tap', TEST], { cwd: RAIZ, encoding: 'utf8', env: entorno });
  const out = String(r.stdout || '');
  const n = (k) => Number((new RegExp('^# ' + k + ' (\\d+)', 'm').exec(out) || [])[1] ?? NaN);
  const caidos = [...out.matchAll(/^not ok \d+ - (.*)$/gm)].map((m) => m[1].slice(0, 110));
  return { exit: r.status, tests: n('tests'), pass: n('pass'), fail: n('fail'), caidos, bytes: out.length };
}

console.log(`POBLACION=${MUTACIONES.length} mutaciones · HEAD=${git('rev-parse', 'HEAD').stdout.trim()}`);
const sucio0 = git('status', '--porcelain').stdout.trim();
if (sucio0) { console.log('ARBOL SUCIO AL EMPEZAR, no se muta:\n' + sucio0); process.exit(2); }
const base = correr();
console.log(`BASE sin mutar: exit=${base.exit} tests=${base.tests} pass=${base.pass} fail=${base.fail} (salida ${base.bytes} B)`);
if (base.exit !== 0 || !(base.tests > 0)) { console.log('BASE NO VERDE o CIEGA: no se muta'); process.exit(2); }

let caen = 0, mudas = 0, ciegas = 0;
for (const m of MUTACIONES) {
  const abs = path.join(RAIZ, m.f);
  const src = fs.readFileSync(abs, 'utf8');
  const veces = src.split(m.de).length - 1;
  if (veces !== 1) { ciegas++; console.log(`\n${m.nombre}\n   CIEGA: el texto a mutar aparece ${veces} veces`); continue; }
  fs.writeFileSync(abs, src.replace(m.de, m.a));
  const numstat = git('diff', '--numstat').stdout.trim().replace(/\s+/g, ' ');
  const r = correr();
  git('restore', '--source=HEAD', '--', m.f);
  const limpio = git('status', '--porcelain').stdout.trim() === '';
  const veredicto = !(r.tests > 0) ? 'CIEGA' : (r.fail > 0 ? 'CAE' : 'MUDA');
  if (veredicto === 'CAE') caen++; else if (veredicto === 'MUDA') mudas++; else ciegas++;
  console.log(`\n${m.nombre}\n   aplicada: ${numstat} · exit=${r.exit} tests=${r.tests} pass=${r.pass} fail=${r.fail} → ${veredicto} · árbol limpio tras restaurar: ${limpio}`);
  for (const c of r.caidos) console.log(`     ✖ ${c}`);
  if (!limpio) { console.log('EL ARBOL NO QUEDO LIMPIO: paro'); process.exit(3); }
}
const fin = correr();
console.log(`\nBASE después: exit=${fin.exit} tests=${fin.tests} pass=${fin.pass} fail=${fin.fail}`);
console.log(`RESULTADO: ${caen} caen · ${mudas} mudas · ${ciegas} ciegas, de ${MUTACIONES.length}`);
console.log(`EXIT=${mudas || ciegas || fin.exit ? 1 : 0}`);
