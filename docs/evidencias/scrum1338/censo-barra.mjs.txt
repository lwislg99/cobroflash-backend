// ── 1 · los gates del servidor ──────────────────────────────────────────────────────────────
const roleOf = (h) => (h && h.__requiredRole) || null;
const rutas = new Map();
const add = (method, p, role) => {
  const k = `${method.toUpperCase()} ${p}`;
  const prev = rutas.get(k);
  if (prev) { prev.role = prev.role || role; return; }
  rutas.set(k, { method: method.toUpperCase(), path: p, role });
};
for (const layer of app.router.stack) {
  if (!layer.route) continue;
  const p = layer.route.path;
  if (typeof p !== 'string' || !p.startsWith('/admin')) continue;
  const role = layer.route.stack.map((s) => roleOf(s.handle)).find(Boolean) || null;
  for (const m of Object.keys(layer.route.methods)) add(m, p, role);
}
for (const mount of getAdminMounts()) {
  const mountRole = mount.gates.map(roleOf).find(Boolean) || null;
  const useRole = mount.router.stack.filter((l) => !l.route).map((l) => roleOf(l.handle)).find(Boolean) || null;
  for (const layer of mount.router.stack) {
    if (!layer.route) continue;
    const routeRole = layer.route.stack.map((s) => roleOf(s.handle)).find(Boolean) || null;
    const rel = layer.route.path === '/' ? '' : layer.route.path;
    for (const m of Object.keys(layer.route.methods)) add(m, mount.prefix + rel, mountRole || useRole || routeRole);
  }
}
const tabla = [...rutas.values()].map((r) => ({
  ...r,
  literales: r.path.split('/').filter((s) => s && !s.startsWith(':')).length,
  re: new RegExp('^' + r.path.split('/').map((s) => (s.startsWith(':') ? '[^/]+' : s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))).join('/') + '/?$'),
}));
function gate(method, url) {
  const p = String(url).replace(/^https?:\/\/[^/]+/, '').replace(/[?#].*$/, '');
  const c = tabla.filter((r) => r.method === method.toUpperCase() && r.re.test(p)).sort((a, b) => b.literales - a.literales);
  return c.length ? c[0] : null;
}
console.log(`POBLACION_RUTAS=${tabla.length} · con rol admin=${tabla.filter((r) => r.role === 'admin').length} · sin rol=${tabla.filter((r) => !r.role).length}`);
console.log(`control gate: GET /admin/cobros -> ${JSON.stringify(gate('GET', '/admin/cobros')?.role)} · GET /admin/customers -> ${JSON.stringify(gate('GET', '/admin/customers')?.role)} · GET /admin/no-existe -> ${JSON.stringify(gate('GET', '/admin/no-existe'))}`);

  };

// ── 3 · montar cada vista como operario, con una red que contesta lo que el gate diría ──────
async function montar(rol, fn, args = []) {
  const peticiones = [];
  const fetch = async (url, opts) => {
    const method = (opts && opts.method) || 'GET';
    const g = gate(method, url);
    const niega = g && g.role === 'admin' && rol !== 'admin';
    peticiones.push({ method, url: String(url).replace(/^https?:\/\/[^/]+/, '').replace(/\?.*$/, ''), ruta: g ? g.path : null, rol: g ? g.role : undefined, status: niega ? 403 : 200 });
    if (niega) {
      return { ok: false, status: 403, headers: { get: () => 'application/json' }, json: async () => ({ error: 'forbidden', required_role: 'admin' }), text: async () => '{"error":"forbidden","required_role":"admin"}', blob: async () => ({}) };
    }
    return { ok: true, status: 200, headers: { get: () => 'application/json' }, json: async () => (/\/admin\/merchant(\?|$)/.test(String(url)) ? { id: 1, name: 'Taller', defaultCurrency: 'EUR', country: 'ES' } : /\/admin\/(products|providers)(\?|$)/.test(String(url)) ? { ok: true, items: [] } : datosDeMuestra(String(url))), text: async () => '', blob: async () => ({}) };
  };
  const b = cargarDashboard(RAIZ, { rol, red: { navigator: { userAgent: 'banco', language: 'es-ES', onLine: true, serviceWorker: { register: async () => ({}) } }, fetch } });
  b.ctx.appMerchantId = 1;
  b.ctx.appDocumentoSuelto = 'no';
  b.ctx.renderAppView = () => {};
  const v = await pintarVista(b, fn, ...args);
  for (let i = 0; i < 20; i++) await new Promise((r) => setImmediate(r));
  let pinta = null;
  if (v.contenedor) {
    const ns = todos(v.contenedor);
    const cls = (n) => String(n.className || '');
    pinta = {
      nodos: ns.length,
      esqueletos: ns.filter((n) => /skeleton|skel\b/.test(cls(n))).length,
      carteles: ns.filter((n) => /error|empty-state|alert/.test(cls(n))).map((n) => `${cls(n)}: ${String(n.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 90)}`).slice(0, 4),
      botones: ns.filter((n) => n.tagName === 'BUTTON').length,
      crearProducto: (() => { const b = ns.find((n) => n.id === 'pf-create-product'); return b ? (b.disabled ? 'DESHABILITADO' : 'pulsable') : null; })(),
    };
  }
  return { peticiones, error: v.error ? String(v.error.message) : null, noMedida: v.noMedida || null, rechazos: v.rechazos || [], pinta };
}
