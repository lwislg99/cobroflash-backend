// ── 1 · los gates del servidor ──────────────────────────────────────────────────────────────
const roleOf = (h) => (h && h.__requiredRole) || null;
const rutas = new Map();
const add = (method, p, role) => {
  const k = `${method.toUpperCase()} ${p}`;
  const prev = rutas.get(k);
  if (prev) { prev.role = prev.role || role; return; }
  rutas.set(k, { method: method.toUpperCase(), path: p, role });
};
for (const layer of app.router.stack) {
  if (!layer.route) continue;
  const p = layer.route.path;
  if (typeof p !== 'string' || !p.startsWith('/admin')) continue;
  const role = layer.route.stack.map((s) => roleOf(s.handle)).find(Boolean) || null;
  for (const m of Object.keys(layer.route.methods)) add(m, p, role);
}
for (const mount of getAdminMounts()) {
  const mountRole = mount.gates.map(roleOf).find(Boolean) || null;
  const useRole = mount.router.stack.filter((l) => !l.route).map((l) => roleOf(l.handle)).find(Boolean) || null;
  for (const layer of mount.router.stack) {
    if (!layer.route) continue;
    const routeRole = layer.route.stack.map((s) => roleOf(s.handle)).find(Boolean) || null;
    const rel = layer.route.path === '/' ? '' : layer.route.path;
    for (const m of Object.keys(layer.route.methods)) add(m, mount.prefix + rel, mountRole || useRole || routeRole);
  }
}
const tabla = [...rutas.values()].map((r) => ({
  ...r,
  literales: r.path.split('/').filter((s) => s && !s.startsWith(':')).length,
  re: new RegExp('^' + r.path.split('/').map((s) => (s.startsWith(':') ? '[^/]+' : s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))).join('/') + '/?$'),
}));
function gate(method, url) {
  const p = String(url).replace(/^https?:\/\/[^/]+/, '').replace(/[?#].*$/, '');
  const c = tabla.filter((r) => r.method === method.toUpperCase() && r.re.test(p)).sort((a, b) => b.literales - a.literales);
  return c.length ? c[0] : null;
}
