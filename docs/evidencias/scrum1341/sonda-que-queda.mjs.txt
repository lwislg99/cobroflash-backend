const ahora = new Date();

// Aplana un valor a { ruta: valor } con rutas del tipo members[Ana].collected.
function aplanar(v, pre, sal) {
  if (Array.isArray(v)) {
    if (v.length === 0) { sal[pre + '[]'] = '(vacía)'; return; }
    v.forEach((x, i) => {
      const clave = x && typeof x === 'object' && 'name' in x ? x.name : String(i);
      aplanar(x, pre + '[' + clave + ']', sal);
    });
    // El ORDEN es un dato aparte: se guarda como una ruta más.
    if (v[0] && typeof v[0] === 'object' && 'name' in v[0]) sal[pre + '.ORDEN'] = v.map((x) => x.name).join(' > ');
  } else if (v && typeof v === 'object') {
    for (const k of Object.keys(v)) aplanar(v[k], pre ? pre + '.' + k : k, sal);
  } else {
    sal[pre] = v === null ? 'null' : String(v);
  }
}

  const p = {}; aplanar(r, '', p); planos[nombre] = p;

console.log('\n── RUTAS QUE NO SE MUEVEN (lo que queda): ' + quietas.length);

// Un campo que se mueve para UNA persona es un campo de dinero para todas: la fila de quien no
